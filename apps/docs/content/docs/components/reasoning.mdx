## Installation

```bash
npx shadcn@latest add @simple-ai/reasoning
```

The CLI also installs shadcn `collapsible`, plus `motion` and `streamdown`.

## Usage

```tsx
import {
  Reasoning,
  ReasoningContent,
  ReasoningTrigger,
} from "@/components/ui/reasoning"

export function Thoughts({ text, isStreaming }: { text: string; isStreaming: boolean }) {
  return (
    <Reasoning isStreaming={isStreaming}>
      <ReasoningTrigger />
      <ReasoningContent>{text}</ReasoningContent>
    </Reasoning>
  )
}
```

Pass `isStreaming` from the last assistant message while `useChat` status is `streaming`. The trigger shows a shimmer while thinking, then “Thought for N seconds”.
